// 1. URL CSV Resmi Google Sheet BUMDesa Lampung Anda (Tab BUM Desa) const GOOGLE_SHEET_CSV_URL = 'https://docs.google.com/spreadsheets/d/e/2PACX-1vRUaww1H2Gp26beZRhRa18rzbJbjKzbziHwUas4XTUQfftg0AIDvBLegfFm23wfyxOWNGgvmTDi3WoZ/pub?gid=1479470316&single=true&output=csv'; // 2. Fungsi Pembantu untuk Konversi Teks Uang/Rupiah ke Angka Murni (Aman dari Crash) function parseRupiah(val) { if (!val) return 0; // Hapus semua karakter non-angka let clean = val.toString().replace(/[^0-9]/g, ''); return parseInt(clean, 10) || 0; } // 3. Fungsi Utama Ambil & Olah Data Google Sheet (Diperbaiki & Kebal Error) function fetchGoogleSheetData() { // Ambil URL dari input box atau gunakan konstanta default const csvUrlInput = document.getElementById('csvUrlInput'); const targetUrl = (csvUrlInput && csvUrlInput.value.trim()) ? csvUrlInput.value.trim() : GOOGLE_SHEET_CSV_URL; updateSyncStatus('Sinkronisasi data...', 'bg-amber-400 animate-pulse'); Papa.parse(targetUrl, { download: true, header: false, // Diset false agar dapat melompati baris judul & header bertingkat skipEmptyLines: true, complete: function(results) { const rawRows = results.data; if (!rawRows || rawRows.length === 0) { console.warn("Data CSV kosong."); updateSyncStatus('Gagal: Data Kosong', 'bg-rose-500'); return; } // Mencari indeks baris tempat Header "Nama BUM Desa" berada let headerIndex = -1; for (let i = 0; i < Math.min(rawRows.length, 10); i++) { if (rawRows[i] && rawRows[i].some(cell => cell && cell.toString().toLowerCase().includes('nama bum desa'))) { headerIndex = i; break; } } // Fallback jika header tidak ditemukan secara otomatis (Default ke baris index 4) if (headerIndex === -1) headerIndex = 4; // Ambil data BUMDesa mulai dari 2 baris setelah header (melompati baris nomor kolom 1,2,3...) const dataRows = rawRows.slice(headerIndex + 2); // Transformasi & Pemetaan Kolom Data const parsedData = dataRows .filter(row => row[10] && row[10].trim() !== '' && !row[10].toString().toLowerCase().includes('kolom')) .map((row, idx) => ({ id: idx + 1, provinsi: row[2] || 'LAMPUNG', kabupaten: row[4] || 'LAMPUNG SELATAN', kecamatan: (row[6] || '-').trim(), desa: (row[8] || '-').trim(), namaBumdes: (row[10] || '-').trim(), tahun: parseInt(row[11]) || 2016, noPerdes: row[12] || '-', statusHukum: (row[13] || 'Belum Berbadan Hukum').trim(), noSertifikatAhu: row[21] || '-', direktur: row[26] || '-', pegawai: parseRupiah(row[32]), modalApbdes: parseRupiah(row[54] || row[43] || 0) })); console.log("Data BUMDesa Terbaca Sukses:", parsedData.length, "baris"); // Update Variabel Global & Panggil Fungsi Sinkronisasi UI masterData = parsedData; if (typeof allBumdesData !== 'undefined') allBumdesData = parsedData; // Panggil fungsi pembuat dropdown & tabel (dengan pemeriksaan ketersediaan fungsi) if (typeof populateKecamatanDropdown === 'function') populateKecamatanDropdown(masterData); if (typeof initFilterDropdowns === 'function') initFilterDropdowns(masterData); if (typeof applyFilters === 'function') applyFilters(); updateSyncStatus('Connected (Google Sheet Live)', 'bg-emerald-400'); }, error: function(err) { console.error("Gagal membaca Google Sheets CSV:", err); updateSyncStatus('Terputus (Error CSV)', 'bg-rose-500'); } }); }
